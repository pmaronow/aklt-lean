def AKLT.Defs.physicalEval : {L N : AKLT.Challenge.Size} →
  AKLT.Challenge.Hilbert L N → (AKLT.Challenge.Site L N → AKLT.Defs.Spinor) → ℂ :=
fun {L N} ψ z => ∑ σ, ψ.ofLp σ * ∏ x, ↑√↑(Nat.choose 4 ↑(σ x)) * z x 0 ^ (4 - ↑(σ x)) * z x 1 ^ ↑(σ x)
def AKLT.Defs.polynomialState : {L N : AKLT.Challenge.Size} →
  AKLT.Challenge.SpinorPolynomial L N → AKLT.Challenge.Hilbert L N :=
fun {L N} p => WithLp.toLp 2 fun σ => p.coeff (AKLT.Challenge.monomialDegree σ) / ∏ x, ↑√↑(Nat.choose 4 ↑(σ x))
def AKLT.Challenge.monomialDegree : {L N : AKLT.Challenge.Size} →
  AKLT.Challenge.Configuration L N → AKLT.Challenge.Site L N × Fin 2 →₀ ℕ :=
fun {L N} σ => Finsupp.equivFunOnFinite.symm fun p => if p.2 = 0 then 4 - ↑(σ p.1) else ↑(σ p.1)
