def AKLT.Defs.MemoryQuotient.Correct : {m q : ℕ} → AKLT.Defs.MemoryQuotient m q → Prop :=
fun {m q} Q =>
  (∀ (a : Fin q), AKLT.Defs.RawWindow (Q.representative a) ∧ Eq.{1} (Q.project (Q.representative a)) a) ∧
    ∀ (w : AKLT.Defs.DirectionWord m),
      AKLT.Defs.RawWindow w →
        ∀ (d : AKLT.Defs.Compass),
          (Eq.{1} (Q.allowed (Q.project w) d) true ↔ AKLT.Defs.WindowAdmissible w d) ∧
            (AKLT.Defs.WindowAdmissible w d →
              Eq.{1} (Q.kind (Q.project w) d) (AKLT.Defs.windowKind w d) ∧
                Eq.{1} (Q.successor (Q.project w) d) (Q.project (AKLT.Defs.windowSuccessor w d)) ∧
                  (Ne.{1} (AKLT.Defs.windowKind w d) AKLT.Defs.StepKind.revisit →
                    Eq.{1} (Q.arrival (Q.project w)) (AKLT.Defs.windowArrival w)))
